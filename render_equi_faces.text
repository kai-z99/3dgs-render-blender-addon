# Blender 3.x — Render ONLY vertical equirectangular strips to disk
# Now auto-sizes strips so the final stitched pano is exactly 2:1 (width = 2 * HEIGHT).
# You choose NUM_STRIPS and HEIGHT; strip widths are computed so their sum is 2*HEIGHT.
# Uses a Cycles panorama camera (Equirectangular) and either crops across X
# or keeps a fixed crop and rotates around Z between strips (your choice).
# Output: strip_000..strip_{N-1} written to OUTPUT_DIR at computed WIDTH_i x HEIGHT.

import bpy
import os
import math
from mathutils import Vector

# =========================
# ==== DEBUG CONTROLS  ====
# =========================
DEBUG = True
PRINT_SAVE_CHECKS = True

def dbg(msg):
    if DEBUG:
        print(f"[EQSTRIP] {msg}", flush=True)

# =========================
# ======   SETTINGS   =====
# =========================
# Geometry / tiling
NUM_STRIPS      = 8            # 1 = full 360 in one shot; N strips each 360/N degrees
HEIGHT          = 2048         # pixel height of each strip (equirect full vertical FOV)
ROTATE_AROUND_Z = True         # If True: rotate Z per strip & render leftmost band.
                               # If False: fixed camera; slide crop window across X.

# I/O and render format
OUTPUT_DIR     = "//eq_strips/"   # relative to .blend unless absolute path
FILE_FMT       = 'PNG'            # 'PNG' (LDR), 'OPEN_EXR' (HDR), or 'HDR' (Radiance)
VIEW_TRANSFORM = 'Standard'       # 'Standard' for LDR, 'Raw' for scene-linear workflows
ENGINE         = 'CYCLES'         # Must be CYCLES for panoramic equirect
ATOMIC_MOVES   = False            # Optional: render to temp and then atomic move

# =========================
# ======  CONSTANTS   =====
# =========================
EXT_MAP = {
    'PNG': '.png', 'OPEN_EXR': '.exr', 'HDR': '.hdr', 'JPEG': '.jpg',
    'TIFF': '.tif','TARGA': '.tga','TARGA_RAW': '.tga'
}
EXT = EXT_MAP.get(FILE_FMT, '.png')
IS_HDRLIKE = FILE_FMT in {'OPEN_EXR','HDR'}
IS_EXR     = FILE_FMT == 'OPEN_EXR'
IS_RAD_HDR = FILE_FMT == 'HDR'
COLORSPACE = 'Linear' if IS_HDRLIKE else 'sRGB'

# Ensure output dirs exist
abs_out = bpy.path.abspath(OUTPUT_DIR)
os.makedirs(abs_out, exist_ok=True)

# Optional temp staging dir for atomic moves
_tmp = os.path.join(abs_out, "_tmp")
if ATOMIC_MOVES:
    os.makedirs(_tmp, exist_ok=True)

# Ensure a camera exists (re-use first camera or create one at 3D cursor)
cam = None
for obj in bpy.data.objects:
    if obj.type == 'CAMERA':
        cam = obj
        break
if cam is None:
    cam_data = bpy.data.cameras.new("EquirectCam")
    cam = bpy.data.objects.new("EquirectCam", cam_data)
    bpy.context.collection.objects.link(cam)
    dbg("Created camera: EquirectCam")

# Position camera at 3D cursor (raise slightly to avoid clipping in some scenes)
cursor = bpy.context.scene.cursor.location
cam.location = Vector((cursor.x, cursor.y, cursor.z + 0.0))
cam.rotation_euler = (0.0, 0.0, 0.0)

# Configure as panoramic equirectangular (Cycles only)
cam.data.type = 'PANO'
if hasattr(cam.data, 'cycles'):
    cam.data.cycles.panorama_type = 'EQUIRECTANGULAR'

# Scene/render setup
scene = bpy.context.scene
scene.camera = cam
scene.render.engine = ENGINE

# Final pano must be exactly 2:1; set total width accordingly.
BASE_W = int(2 * HEIGHT)  # final stitched width
scene.render.resolution_x = BASE_W
scene.render.resolution_y = HEIGHT
scene.render.resolution_percentage = 100

scene.render.image_settings.file_format = FILE_FMT
scene.render.image_settings.color_mode = 'RGB' if IS_RAD_HDR else 'RGBA'
# Guard: some Blender configs need view_settings access via display settings
try:
    scene.view_settings.view_transform = VIEW_TRANSFORM
except Exception:
    pass
scene.render.use_file_extension = True
scene.render.use_overwrite = True
scene.render.film_transparent = False

# Crop/border render to produce vertical strips
scene.render.use_border = True
scene.render.use_crop_to_border = True  # output size == cropped area

# HDR specifics
if IS_EXR:
    scene.render.image_settings.color_depth = '32'
    scene.render.image_settings.exr_codec = 'ZIP'
elif IS_RAD_HDR:
    scene.render.image_settings.color_depth = '32'  # Radiance HDR is RGB-only

# Helpers

def set_yaw_deg(z_degrees: float):
    # Keep X/Y (pitch/roll) set for level horizon; yaw around Z
    cam.rotation_euler[0] = math.radians(90)
    cam.rotation_euler[1] = math.radians(0)
    cam.rotation_euler[2] = math.radians(z_degrees)

def set_border(min_x: float, max_x: float):
    # min/max in [0,1] across the full equirectangular width
    scene.render.border_min_x = min_x
    scene.render.border_max_x = max_x
    scene.render.border_min_y = 0.0
    scene.render.border_max_y = 1.0

def pixel_edges(total_w: int, n: int):
    """Return integer pixel edges [e0..en] that partition total_w into n strips
    whose widths differ by at most 1 px and sum exactly to total_w."""
    # e_i = floor(i * total_w / n)
    edges = [(i * total_w) // n for i in range(n + 1)]
    return edges

# Log base
dbg(f"START | strips={NUM_STRIPS} height={HEIGHT} -> pano={BASE_W}x{HEIGHT} (2:1) engine={ENGINE} fmt={FILE_FMT} view={VIEW_TRANSFORM}")
dbg(f"HDRLIKE={IS_HDRLIKE} EXR={IS_EXR} RAD_HDR={IS_RAD_HDR} COLORSPACE={COLORSPACE} EXT={EXT}")

# Precompute exact pixel edges and per-strip widths
edges_px = pixel_edges(BASE_W, NUM_STRIPS)
widths_px = [edges_px[i+1] - edges_px[i] for i in range(NUM_STRIPS)]
dbg(f"Per-strip widths (px): {widths_px} (sum={sum(widths_px)})")

# Render N strips
for i in range(NUM_STRIPS):
    if ROTATE_AROUND_Z:
        # Rotate Z by i * 360/N; render the leftmost band with exact pixel width
        set_yaw_deg(i * (360.0 / NUM_STRIPS))
        w_px = widths_px[i]
        min_x = 0.0
        max_x = w_px / BASE_W
        set_border(min_x, max_x)
    else:
        # Fixed camera; slide the crop window to exact pixel-aligned span
        set_yaw_deg(0.0)
        min_x = edges_px[i] / BASE_W
        max_x = edges_px[i+1] / BASE_W
        set_border(min_x, max_x)

    base_name = f"strip_{i:03d}"
    if ATOMIC_MOVES:
        scene.render.filepath = bpy.path.relpath(os.path.join(_tmp, base_name))
        bpy.ops.render.render(write_still=True)
        src = os.path.join(_tmp, base_name + EXT)
        dst = os.path.join(abs_out, base_name + EXT)
        os.replace(src, dst)
        out_path = dst
    else:
        scene.render.filepath = OUTPUT_DIR + base_name
        bpy.ops.render.render(write_still=True)
        out_path = os.path.join(abs_out, base_name + EXT)

    size = -1
    try:
        size = os.path.getsize(out_path)
    except Exception:
        pass
    if PRINT_SAVE_CHECKS:
        dbg(f"Rendered {base_name} -> {bpy.path.abspath(out_path)} "
            f"(bytes={size})")

print("Equirectangular strips written to:", abs_out)
dbg("DONE.")
